# their headers (include/unicode) and pkg-config files (icu-uc, icu-i18n) into the POSIX sysroot.
# tools/ports/icu/build.sh runs it with tools/onyx-toolchain.cmake; the data is made by gen-data.sh.
#
# ICU's own build (autoconf) does not know an Onyx host and wants a host build next to it for the data;
# the data is prebuilt here (as ICU's own source release ships its icudt78l.dat), so the libraries are
# plain lists of sources: every .cpp of common/ and i18n/, as ICU's Makefiles list them.

cmake_minimum_required(VERSION 3.20)
project(onyx_icu VERSION 78.3 LANGUAGES C CXX ASM)

set(ICU_SRC "" CACHE PATH "third_party/icu-78.3/source")
set(ICU_DAT "${ICU_SRC}/data/in/icudt78l.dat" CACHE FILEPATH "the data file linked in")
if(NOT EXISTS "${ICU_SRC}/common/putil.cpp")
	message(FATAL_ERROR "ICU_SRC: no ICU sources at '${ICU_SRC}'")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)		# gnu++17: newlib's POSIX declarations

file(GLOB ICU_UC_SOURCES "${ICU_SRC}/common/*.cpp")
file(GLOB ICU_I18N_SOURCES "${ICU_SRC}/i18n/*.cpp")

# What ICU's configure would decide for a POSIX host, written out: a static library (no DLL import
# attributes), no plug-ins (no dlopen), newlib's time-zone variables, no mmap of data files (the data is
# linked in; a package opened with udata_open reads through stdio), the default code page UTF-8.
set(ICU_DEFS
	U_ATTRIBUTE_DEPRECATED=
	U_STATIC_IMPLEMENTATION
	U_ENABLE_DYLOAD=0
	U_HAVE_ELF_H=1
	U_HAVE_MMAP=0
	U_TIMEZONE=_timezone
	U_TZNAME=_tzname
	U_CHARSET_IS_UTF8=1
	U_HAVE_STRTOD_L=0
	U_HAVE_XLOCALE_H=0
	"U_ICU_DATA_DEFAULT_DIR=\"SD:/res/icu\"")

add_library(icuuc STATIC ${ICU_UC_SOURCES})
target_compile_definitions(icuuc PRIVATE U_COMMON_IMPLEMENTATION ${ICU_DEFS})
target_include_directories(icuuc PRIVATE "${ICU_SRC}/common")

add_library(icui18n STATIC ${ICU_I18N_SOURCES})
target_compile_definitions(icui18n PRIVATE U_I18N_IMPLEMENTATION ${ICU_DEFS})
target_include_directories(icui18n PRIVATE "${ICU_SRC}/i18n" "${ICU_SRC}/common")

# the data: the .dat file's bytes as the symbol icudt78_dat (what ICU's genccode writes, as one .incbin)
set(ICU_DATA_S "${CMAKE_CURRENT_BINARY_DIR}/icudt78l_dat.S")
file(WRITE "${ICU_DATA_S}" "/* generated by tools/ports/icu/CMakeLists.txt: ${ICU_DAT} */
	.section .rodata.icudt78_dat,\"a\"
	.balign 16
	.globl icudt78_dat
	.type icudt78_dat, %object
icudt78_dat:
	.incbin \"${ICU_DAT}\"
	.size icudt78_dat, . - icudt78_dat
	.section .note.GNU-stack,\"\",%progbits
")
set_source_files_properties("${ICU_DATA_S}" PROPERTIES OBJECT_DEPENDS "${ICU_DAT}")
add_library(icudata STATIC "${ICU_DATA_S}")

foreach(lib icuuc icui18n)
	target_compile_options(${lib} PRIVATE -O2 -w)
endforeach()

install(TARGETS icuuc icui18n icudata ARCHIVE DESTINATION lib)
file(GLOB ICU_UC_HEADERS "${ICU_SRC}/common/unicode/*.h")
file(GLOB ICU_I18N_HEADERS "${ICU_SRC}/i18n/unicode/*.h")
install(FILES ${ICU_UC_HEADERS} ${ICU_I18N_HEADERS} DESTINATION include/unicode)

# pkg-config: as ICU's (icu-uc carries the data; static: the C++ runtime is the toolchain's)
foreach(pc icu-uc icu-i18n)
	if(pc STREQUAL "icu-uc")
		set(PC_DESC "International Components for Unicode: Common and Data libraries")
		set(PC_LIBS "-licuuc -licudata")
		set(PC_REQ "")
	else()
		set(PC_DESC "International Components for Unicode: Internationalization library")
		set(PC_LIBS "-licui18n")
		set(PC_REQ "Requires: icu-uc\n")
	endif()
	file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${pc}.pc" "prefix=${CMAKE_INSTALL_PREFIX}
libdir=\${prefix}/lib
includedir=\${prefix}/include

Name: ${pc}
Description: ${PC_DESC}
Version: 78.3
${PC_REQ}Libs: -L\${libdir} ${PC_LIBS}
Libs.private: -lstdc++ -lm
Cflags: -I\${includedir}
")
	install(FILES "${CMAKE_CURRENT_BINARY_DIR}/${pc}.pc" DESTINATION lib/pkgconfig)
endforeach()
